    // DECODIFICADOR Y SOPORTE DE ARCHIVOS (INCLUIDO E-PUB REAL)
    async function handleFileImport(event) {
        const file = event.target.files[0];
        if (!file) return;

        const ext = file.name.split('.').pop().toLowerCase();
        let title = file.name.replace(/\.[^/.]+$/, "");
        let content = "";
        let format = ext.toUpperCase();

        showLoader("Cargando y procesando libro...");

        try {
            if (ext === 'txt') {
                content = await file.text();
            } 
            else if (ext === 'html' || ext === 'htm') {
                const rawHtml = await file.text();
                const doc = new DOMParser().parseFromString(rawHtml, 'text/html');
                content = doc.body.innerText || doc.body.textContent;
            } 
            else if (ext === 'pdf') {
                const arrayBuffer = await file.arrayBuffer();
                const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
                let text = "";
                for (let i = 1; i <= pdf.numPages; i++) {
                    const page = await pdf.getPage(i);
                    const contentObj = await page.getTextContent();
                    text += contentObj.items.map(item => item.str).join(" ") + "\n\n";
                }
                content = text;
            } 

                
            // LÓGICA DE PROCESAMIENTO E-PUB REAL CON JSZIP
           
            else if (ext === 'epub') {

                const arrayBuffer = await file.arrayBuffer();
                const zip = await JSZip.loadAsync(arrayBuffer);

                // Buscar TODOS los posibles capítulos/textos
                const textFiles = Object.keys(zip.files)
                    .filter(path =>
                        path.toLowerCase().endsWith('.xhtml') ||
                        path.toLowerCase().endsWith('.html') ||
                        path.toLowerCase().endsWith('.htm') ||
                        path.toLowerCase().endsWith('.xml')
                )
                    .sort();

                if (textFiles.length === 0) {
        throw new Error("Este EPUB no contiene páginas legibles.");
    }

    let fullEpubText = "";

    for (const filePath of textFiles) {

        try {

            const rawContent = await zip.files[filePath].async("text");

            // Parsear como XHTML/XML
            const parser = new DOMParser();
            const doc = parser.parseFromString(rawContent, "application/xhtml+xml");

            // Eliminar elementos innecesarios
            doc.querySelectorAll("script, style, nav").forEach(el => el.remove());

            // Extraer texto
            let text = doc.documentElement.textContent || "";

            // Limpiar espacios excesivos
            text = text
                .replace(/\s+/g, " ")
                .replace(/\n\s*\n/g, "\n\n")
                .trim();

            // Evitar agregar basura vacía
            if (text.length > 80) {
                fullEpubText += text + "\n\n";
            }

        } catch (sectionError) {
            console.warn("Error leyendo sección EPUB:", filePath);
        }
    }

    if (!fullEpubText.trim()) {
        throw new Error("No se pudo extraer texto útil del EPUB.");
    }

    content = fullEpubText;
}
                

                
            else {
                showToast("Formato no soportado.");
                return;
            }

            if (!content || content.trim().length === 0) {
                throw new Error("El archivo no contiene texto legible.");
            }

            // Guardar en la biblioteca local
            const colors = ["#7c2d12", "#115e59", "#1e3a8a", "#581c87", "#881337"];
            const randomColor = colors[Math.floor(Math.random() * colors.length)];

            library.push({
                id: "b_" + Date.now(),
                title: title,
                author: "Importado Local",
                shelf: currentShelf !== "Todos" ? currentShelf : "",
                color: randomColor,
                format: format,
                content: content
            });

            saveState();
            renderBooks();
            closeAddBookModal();
            
            hideLoader();
            showSuccess("Cargado con éxito");

        } catch (err) {
            console.error(err);
            hideLoader();
            showToast("Error al decodificar: " + err.message);
        }
    }

    // VISOR E-READER MEJORADO (SIN MARCOS EXTRAÑOS NI BOTONES DUPLICADOS)
    function readBook(id) {
        activeBook = library.find(b => b.id === id);
        if (!activeBook) return;

        document.getElementById("reader-book-title").textContent = activeBook.title;
        
        // Renderizar el texto puro e inteligente
        const container = document.getElementById("reader-text-container");
        container.innerHTML = activeBook.content.replace(/\n/g, '<br>');

        applyReaderSettings();
        switchView('reader');
    }

    function applyReaderSettings() {
        const scroller = document.getElementById("reader-scroller");
        const container = document.getElementById("reader-text-container");
        
        // Temas de Fondo
        if (readerConfig.theme === 'light') {
            scroller.className = "reader-vertical flex-1 p-6 md:p-12 bg-white text-slate-900";
        } else if (readerConfig.theme === 'sepia') {
            scroller.className = "reader-vertical flex-1 p-6 md:p-12 bg-[#f4ecd8] text-[#5b4636]";
        } else {
            scroller.className = "reader-vertical flex-1 p-6 md:p-12 bg-slate-950 text-slate-200";
        }

        // Modo Columnas (2 Páginas) vs Vertical
        if (readerConfig.viewMode === 'pages' && window.innerWidth >= 768) {
            container.className = "book-columns max-w-6xl mx-auto leading-relaxed text-justify text-base";
        } else {
            container.className = "reader-vertical-content max-w-4xl mx-auto leading-relaxed text-justify text-base";
        }
    }

    function setPCViewMode(mode) {
        readerConfig.viewMode = mode;
        document.getElementById("btn-view-pages").className = mode === 'pages' ? "px-2 py-0.5 bg-teal-500 text-slate-950 rounded text-[10px] font-bold" : "px-2 py-0.5 text-slate-400 rounded text-[10px] font-bold";
        document.getElementById("btn-view-vertical").className = mode === 'vertical' ? "px-2 py-0.5 bg-teal-500 text-slate-950 rounded text-[10px] font-bold" : "px-2 py-0.5 text-slate-400 rounded text-[10px] font-bold";
        applyReaderSettings();
    }

    function setReaderTheme(theme) {
        readerConfig.theme = theme;
        applyReaderSettings();
    }

    // Maximizar sin desordenar la interfaz superior homogénea
    function toggleMaximizeReader() {

    readerConfig.maximized = !readerConfig.maximized;

    const mainSection = document.getElementById("view-reader");
    const btn = document.getElementById("btn-maximize-main");

    // BOTÓN SALIR
    const exitBtn = document.querySelector('button[onclick="exitReader()"]');

    if (readerConfig.maximized) {

        // ACTIVAR MODO PANTALLA COMPLETA
        mainSection.classList.add( "fixed", "inset-0", "z-50", "bg-slate-950" );

        // BOTÓN MODERNO
        btn.innerHTML = "← Regresar"; btn.classList.remove("text-teal-400"); btn.classList.add("text-amber-400");

        // OCULTAR BOTÓN SALIR
        if (exitBtn) { exitBtn.style.display = "none";
        }
    } else {

        // DESACTIVAR PANTALLA COMPLETA
        mainSection.classList.remove( "fixed", "inset-0", "z-50", "bg-slate-950" );

        // RESTAURAR BOTÓN
        btn.innerHTML = "🔲 Maximizar";
        btn.classList.remove("text-amber-400");
        btn.classList.add("text-teal-400");

        // MOSTRAR BOTÓN SALIR
        if (exitBtn) {exitBtn.style.display = "flex";
        }
    }

    applyReaderSettings();
}

    function exitReader() {
        if (readerConfig.maximized) {
            toggleMaximizeReader();
        }
        switchView('library');
    }
